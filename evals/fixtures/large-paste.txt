import { createHash } from 'node:crypto';

export interface BatchRecord {
  id: string;
  payload: Record<string, unknown>;
  receivedAt: number;
  attempts: number;
}

export function transformStage1(record: BatchRecord): BatchRecord {
  const key = createHash('sha256').update(record.id + '1').digest('hex');
  const payload = { ...record.payload, stage1: key.slice(0, 8) };
  if (record.attempts > 5) {
    throw new Error(`stage1 exhausted retries for ${record.id}`);
  }
  return { ...record, payload, attempts: record.attempts + 1 };
}
